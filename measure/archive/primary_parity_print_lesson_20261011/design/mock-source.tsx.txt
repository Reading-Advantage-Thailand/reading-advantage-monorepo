    <Scene place="library" className="min-h-screen px-4 py-6">
      <div className="mx-auto flex w-full max-w-4xl flex-col gap-4">
        <div className="flex flex-wrap items-center gap-3"><Sign>{t.title}</Sign></div>
        <Panel className="w-full" data-print-page>
          <div className="flex items-center gap-3">
            <Sprite src={HOST} size={72} className="shrink-0" />
            <Bubble>{state === "ready" ? t.ready : state === "loading" ? t.wait : t.notYet}</Bubble>
          </div>
          {state === "none" ? (
            <div role="alert" className="flex flex-col gap-2"><h1 className="text-xl font-bold">{t.notReadyPage}</h1><p>{t.notReadyBody}</p></div>
          ) : (
            <>
              <h1 className="sr-only">{t.title}</h1>
              <div data-print-frame><Sheet /></div>
              <p className="text-sm" style={{ color: "var(--cq-ink-soft, inherit)" }}>{t.hint}</p>
            </>
          )}
          <div className="flex flex-wrap items-center gap-3">
            {state === "none" ? null : state === "loading"

              : <RpgButton tone="gold"><PrinterIcon aria-hidden="true" className="size-5" />{t.printNow}</RpgButton>}
            <RpgLink tone="iron" href="#"><ArrowLeftIcon aria-hidden="true" className="size-5" />{t.backLesson}</RpgLink>

    </Scene>
